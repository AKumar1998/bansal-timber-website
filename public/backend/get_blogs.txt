<?php
// get_blogs.php

header("Content-Type: application/json");

// Database credentials
$host = "localhost"; // usually localhost on cPanel
$db_name = "your_database_name"; // replace with your database name
$username = "your_db_username"; // replace with your DB username
$password = "your_db_password"; // replace with your DB password

// Connect to MySQL
$conn = new mysqli($host, $username, $password, $db_name);

// Check connection
if ($conn->connect_error) {
    http_response_code(500);
    echo json_encode(["error" => "Database connection failed: " . $conn->connect_error]);
    exit;
}

// Fetch blogs
$sql = "SELECT id, title, slug, excerpt, image, author_id, created_at, updated_at FROM blogs ORDER BY created_at DESC LIMIT 6";
$result = $conn->query($sql);

$blogs = [];

if ($result) {
    while ($row = $result->fetch_assoc()) {
        $blogs[] = $row;
    }
}

echo json_encode($blogs);

// Close connection
$conn->close();
?>
